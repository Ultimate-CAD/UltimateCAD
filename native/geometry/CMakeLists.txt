cmake_minimum_required(VERSION 3.24)
project(layerling_geometry VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR ON CACHE BOOL "" FORCE)
set(MANIFOLD_USE_BUILTIN_TBB ON CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
FetchContent_Declare(manifold GIT_REPOSITORY https://github.com/elalish/manifold.git GIT_TAG ce50d78021d64507f89e8c9fc2c2e51018117857)
FetchContent_Declare(libigl GIT_REPOSITORY https://github.com/libigl/libigl.git GIT_TAG 40e7900ccbd767f1f360e0eb10f0f1a6432e0993)
FetchContent_Declare(eigen GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git GIT_TAG 3147391d946bb4b6c68edd901f2add6ac1f31f8c)
FetchContent_Declare(TBB GIT_REPOSITORY https://github.com/oneapi-src/oneTBB.git GIT_TAG f1862f38f83568d96e814e469ab61f88336cc595)
FetchContent_MakeAvailable(manifold)
# Only the MPL-2.0 geometry headers are used; no optional libigl/CGAL modules.
foreach(dep libigl eigen)
  FetchContent_GetProperties(${dep})
  if(NOT ${dep}_POPULATED)
    FetchContent_Populate(${dep})
  endif()
endforeach()
add_executable(layerling-geometry main.cpp)
target_include_directories(layerling-geometry PRIVATE ${libigl_SOURCE_DIR}/include ${eigen_SOURCE_DIR})
target_link_libraries(layerling-geometry PRIVATE manifold::manifold TBB::tbb)
target_compile_definitions(layerling-geometry PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN IGL_PARALLEL_FOR_FORCE_SERIAL EIGEN_MPL2_ONLY)
if(MSVC)
  target_compile_options(layerling-geometry PRIVATE /O2 /EHsc /bigobj)
endif()
